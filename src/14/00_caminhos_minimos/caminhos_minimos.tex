\begin{frame}{Caminho mínimo}
	\begin{columns}[c]
		\begin{column}{0.46\textwidth}
			\footnotesize
			\begin{itemize}
				\item \textbf{peso de um caminho}: soma dos pesos das arestas;
				\item \textbf{caminho mínimo} de $s$ a $v$: caminho de menor peso entre
				      eles;
				\item \textbf{distância} de $s$ a $v$: o peso do caminho mínimo.
			\end{itemize}
			\vspace{0.2cm}
			Em $G$, de 0 a 6:
			\begin{itemize}
				\item 0-1-3-6: $4 + 2 + 9 = 15$;
				\item 0-5-4-6: $7 + 1 + 8 = 16$;
				\item 0-2-4-6: $3 + 6 + 8 = 17$.
			\end{itemize}
		\end{column}

\begin{frame}{A busca em largura não olha os pesos}
	\begin{columns}[c]

\begin{frame}[fragile]{Relaxação de uma aresta}
	\footnotesize
	O vetor \texttt{dist} guarda o peso do melhor caminho conhecido até cada
	vértice, e o vetor \texttt{anterior}, o vértice que vem antes dele nesse
	caminho.
	\begin{center}
		\scalebox{0.72}{\relaxacao}
	\end{center}
	\begin{minted}[fontsize=\scriptsize]{go}
if dist[v]+a.Peso < dist[w] { // passar por v melhora o caminho até w?
    dist[w] = dist[v] + a.Peso
    anterior[w] = v
}
\end{minted}
\end{frame}

\section{Dijkstra}

\begin{frame}{Algoritmo de Dijkstra}
	\begin{enumerate}
		\item Comece com \texttt{dist[s]} = 0, as outras distâncias infinitas e
		      $s$ na fila de prioridade;
		\item Retire da fila o candidato de menor distância;
		\item Se o vértice já está na árvore, descarte o candidato;
		\item Senão, coloque o vértice na árvore e relaxe as arestas dele. Cada
		      melhora põe um novo candidato na fila;
		\item Repita até a fila esvaziar.
	\end{enumerate}
	\vspace{0.3cm}
	\footnotesize
	Um vértice pode ter vários candidatos na fila, um para cada melhora da
	distância. Só o menor deles coloca o vértice na árvore.
\end{frame}

\begin{frame}[fragile]{Dijkstra em Go: candidatos e início}
	\begin{minted}[fontsize=\scriptsize]{go}
// Candidato é um vértice que pode entrar na árvore, com a distância da
// origem até ele pelo caminho encontrado até aqui.
type Candidato struct {
    V    int
    Dist float64
}

func Dijkstra(g *GrafoP, s int) (dist []float64, anterior []int) {
    dist = make([]float64, g.V())
    anterior = make([]int, g.V())
    for v := range dist {
        dist[v] = math.Inf(1) // ainda sem caminho
        anterior[v] = -1
    }
    naArvore := make([]bool, g.V())
    fila := &FilaCandidatos{}
\end{minted}
	\footnotesize
	\texttt{FilaCandidatos} tem a forma de \texttt{FilaArestas} do Prim, com
	\texttt{Dist} como prioridade.
\end{frame}

\begin{frame}[fragile]{Dijkstra em Go: laço principal}
	\begin{minted}[fontsize=\scriptsize]{go}
    dist[s] = 0
    heap.Push(fila, Candidato{s, 0})
    for fila.Len() > 0 {
        c := heap.Pop(fila).(Candidato)
        v := c.V
        if naArvore[v] {
            continue // v já entrou por um caminho menor
        }
        naArvore[v] = true
        for _, a := range g.adj[v] {
            w := a.Outro(v)
            if !naArvore[w] && dist[v]+a.Peso < dist[w] {
                dist[w] = dist[v] + a.Peso
                anterior[w] = v
                heap.Push(fila, Candidato{w, dist[w]})
            }
        }
    }
    return dist, anterior
}
\end{minted}
\end{frame}

\begin{frame}{Por que o menor candidato já é a distância}
	\begin{itemize}
		\item o candidato $v$ tem a menor distância da fila, $d$;
		\item qualquer outro caminho de $s$ a $v$ precisa sair da árvore por alguma
		      aresta $x$-$y$, com $x$ na árvore e $y$ fora;
		\item até $y$, esse caminho pesa pelo menos \texttt{dist[y]}, e o candidato
		      de $y$ na fila tem distância maior ou igual a $d$;
		\item de $y$ até $v$, os pesos não são negativos, e o caminho inteiro pesa
		      pelo menos $d$.
	\end{itemize}
	\vspace{0.3cm}
	\footnotesize
	Em $G$, a partir de 0, o vértice 3 sai da fila com distância 6. Os outros
	candidatos têm distância 7 ou mais, e nenhum caminho que passe por eles chega
	ao 3 com peso menor.
\end{frame}

\begin{frame}{Pesos negativos}
	\begin{columns}[c]
		\begin{column}{0.58\textwidth}
			\footnotesize
			\begin{itemize}
				\item o vértice 1 entra na árvore com distância 2;
				\item depois, o vértice 2 entra com distância 3, e a aresta 2-1 daria
				      $3 + (-2) = 1$;
				\item o 1 já está na árvore, e a distância fica errada: 2 em vez de 1.
			\end{itemize}
			\vspace{0.2cm}
			Em grafo não dirigido, uma aresta negativa é um ciclo: ir e voltar por
			ela reduz o peso sem limite. Pesos negativos pedem outro algoritmo, como
			o de Bellman-Ford, que não é tratado nesta disciplina.
		\end{column}
		\begin{column}{0.40\textwidth}
			\begin{center}
				\scalebox{1.1}{\pesoNegativo}
			\end{center}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Custo do Dijkstra}
	\begin{itemize}
		\item uma aresta só é relaxada com sucesso quando o primeiro dos seus
		      extremos entra na árvore;
		\item a fila recebe no máximo $E + 1$ candidatos, e cada operação custa
		      $O(\log E)$;
		\item total: $O(E \log E)$, igual a $O(E \log V)$, porque $\log E \le 2 \log
			      V$;
		\item em grafo denso, um vetor no lugar da fila dá $O(V^2)$, sem heap.
	\end{itemize}
	\vspace{0.3cm}
	\footnotesize
	Com todas as arestas de mesmo peso, a busca em largura encontra o caminho
	mínimo em $O(V + E)$, sem fila de prioridade.
\end{frame}
